package com.example.aman;

import android.content.Context;
import android.content.Intent;
import android.content.SharedPreferences;
import android.os.Bundle;
import android.text.TextUtils;
import android.util.SparseBooleanArray;
import android.widget.ArrayAdapter;
import android.widget.Button;
import android.widget.EditText;
import android.widget.ListView;
import android.widget.RadioButton;
import android.widget.RadioGroup;
import android.widget.Toast;

import androidx.appcompat.app.AppCompatActivity;

import java.util.ArrayList;
import java.util.Calendar;

public class MainActivity extends AppCompatActivity {

    private RadioGroup foodPreferenceRadioGroup;
    private EditText nameEditText;
    private RadioGroup genderRadioGroup;
    private EditText ageEditText;
    private ListView buffetListView;

    private int customerCounter;
    private SharedPreferences sharedPreferences;

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_main);

        sharedPreferences = getSharedPreferences("customer_data", Context.MODE_PRIVATE);
        customerCounter = sharedPreferences.getInt("customerCounter", 1);

        foodPreferenceRadioGroup = findViewById(R.id.foodPreferenceRadioGroup);
        nameEditText = findViewById(R.id.nameEditText);
        genderRadioGroup = findViewById(R.id.genderRadioGroup);
        ageEditText = findViewById(R.id.ageEditText);
        buffetListView = findViewById(R.id.buffetListView);
        Button registerButton = findViewById(R.id.registerButton);

        ArrayAdapter<CharSequence> adapter = ArrayAdapter.createFromResource(this,
                R.array.buffet_items, android.R.layout.simple_list_item_multiple_choice);
        buffetListView.setAdapter(adapter);

        registerButton.setOnClickListener(v -> registerCustomer());
    }

    private void registerCustomer() {
        if (foodPreferenceRadioGroup.getCheckedRadioButtonId() == -1) {
            Toast.makeText(this, "Please select a food preference", Toast.LENGTH_SHORT).show();
            return;
        }
        String foodPreference = ((RadioButton) findViewById(foodPreferenceRadioGroup.getCheckedRadioButtonId())).getText().toString();

        String name = nameEditText.getText().toString().trim();
        if (TextUtils.isEmpty(name)) {
            nameEditText.setError("Name is required");
            nameEditText.requestFocus();
            return;
        }

        if (genderRadioGroup.getCheckedRadioButtonId() == -1) {
            Toast.makeText(this, "Please select a gender", Toast.LENGTH_SHORT).show();
            return;
        }
        String gender = ((RadioButton) findViewById(genderRadioGroup.getCheckedRadioButtonId())).getText().toString();

        String age = ageEditText.getText().toString().trim();
        if (TextUtils.isEmpty(age)) {
            ageEditText.setError("Age is required");
            ageEditText.requestFocus();
            return;
        }

        ArrayList<String> selectedFoodItems = new ArrayList<>();
        SparseBooleanArray checked = buffetListView.getCheckedItemPositions();
        for (int i = 0; i < buffetListView.getAdapter().getCount(); i++) {
            if (checked.get(i)) {
                selectedFoodItems.add((String) buffetListView.getAdapter().getItem(i));
            }
        }

        if (selectedFoodItems.isEmpty()) {
            Toast.makeText(this, "Please select at least one buffet item", Toast.LENGTH_SHORT).show();
            return;
        }

        String customerId = generateCustomerId(foodPreference);

        SharedPreferences.Editor editor = sharedPreferences.edit();
        editor.putInt("customerCounter", customerCounter);
        editor.apply();

        Intent intent = new Intent(MainActivity.this, DisplayCustomerActivity.class);
        intent.putExtra("foodPreference", foodPreference);
        intent.putExtra("name", name);
        intent.putExtra("gender", gender);
        intent.putExtra("age", age);
        intent.putStringArrayListExtra("selectedFoodItems", selectedFoodItems);
        intent.putExtra("customerId", customerId);
        startActivity(intent);
    }

    private String generateCustomerId(String foodPreference) {
        Calendar calendar = Calendar.getInstance();
        int year = calendar.get(Calendar.YEAR);
        int month = calendar.get(Calendar.MONTH) + 1;
        int day = calendar.get(Calendar.DAY_OF_MONTH);

        String foodPrefCode = foodPreference.equals("Vegetarian") ? "Veg" : "NonVeg";

        String id = foodPrefCode + "/" + year + "/Buffet/" + day + "/" + month + "/" + (year % 100) + "/A" + customerCounter;
        customerCounter++;
        return id;
    }
}

package com.example.aman;

import android.content.Intent;
import android.os.Bundle;
import android.widget.TextView;

import androidx.appcompat.app.AppCompatActivity;

import java.util.ArrayList;

public class DisplayCustomerActivity extends AppCompatActivity {

    @Override
    protected void onCreate(Bundle savedInstanceState) {
        super.onCreate(savedInstanceState);
        setContentView(R.layout.activity_display_customer);

        TextView customerInfoTextView = findViewById(R.id.customerInfoTextView);

        Intent intent = getIntent();
        String foodPreference = intent.getStringExtra("foodPreference") != null ? intent.getStringExtra("foodPreference") : "N/A";
        String name = intent.getStringExtra("name") != null ? intent.getStringExtra("name") : "N/A";
        String gender = intent.getStringExtra("gender") != null ? intent.getStringExtra("gender") : "N/A";
        String age = intent.getStringExtra("age") != null ? intent.getStringExtra("age") : "N/A";
        ArrayList<String> selectedFoodItems = intent.getStringArrayListExtra("selectedFoodItems");
        String customerId = intent.getStringExtra("customerId") != null ? intent.getStringExtra("customerId") : "N/A";

        StringBuilder sb = new StringBuilder();
        sb.append("Customer ID: ").append(customerId).append("\n\n");
        sb.append("Name: ").append(name).append("\n");
        sb.append("Gender: ").append(gender).append("\n");
        sb.append("Age: ").append(age).append("\n");
        sb.append("Food Preference: ").append(foodPreference).append("\n");
        sb.append("Ordered Food Items:\n");
        if (selectedFoodItems != null) {
            for (String item : selectedFoodItems) {
                sb.append("- ").append(item).append("\n");
            }
        }

        customerInfoTextView.setText(sb.toString());
    }
}
